'use client';

import React, { useRef, useState } from 'react';
import Webcam from 'react-webcam';
import Image from 'next/image';
import { createWorker } from 'tesseract.js';

const Scanner = () => {
  const webcamRef = useRef<Webcam | null>(null);
  const [image, setImage] = useState<string | null>(null);
  const [category, setCategory] = useState<string>('');
  const [loading, setLoading] = useState<boolean>(false);
  const [error, setError] = useState<string>('');
  const [ocrText, setOcrText] = useState<string>('');
  const [useBackCamera, setUseBackCamera] = useState<boolean>(true);

  const videoConstraints = {
    facingMode: useBackCamera ? { exact: 'environment' } : 'user',
  };

  const captureImage = () => {
    if (!webcamRef.current) return;
    const imageSrc = webcamRef.current.getScreenshot();
    setImage(imageSrc || null);
  };

  const preprocessImage = (image: string): string => {
    // Preprocessing logic using Canvas to convert the image to grayscale
    const canvas = document.createElement('canvas');
    const img = new window.Image();
    img.src = image;

    const processedImage = new Promise<string>((resolve) => {
      img.onload = () => {
        canvas.width = img.width;
        canvas.height = img.height;
        const ctx = canvas.getContext('2d');
        if (ctx) {
          ctx.drawImage(img, 0, 0);
          const imageData = ctx.getImageData(0, 0, img.width, img.height);

          // Convert to grayscale
          for (let i = 0; i < imageData.data.length; i += 4) {
            const avg =
              0.299 * imageData.data[i] +
              0.587 * imageData.data[i + 1] +
              0.114 * imageData.data[i + 2];
            imageData.data[i] = avg; // Red
            imageData.data[i + 1] = avg; // Green
            imageData.data[i + 2] = avg; // Blue
          }

          ctx.putImageData(imageData, 0, 0);
        }
        resolve(canvas.toDataURL());
      };
    });

    return processedImage as unknown as string;
  };

  const postprocessText = (text: string): string => {
    const cleanedText = text
      .replace(/\s+/g, ' ') // Remove extra spaces
      .trim();
    return cleanedText;
  };

  const categorizeExpense = (text: string): string => {
    const categories: { [key: string]: string[] } = {
      Food: ['restaurant', 'coffee', 'burger', 'pizza', 'meal', 'food'],
      Transportation: ['uber', 'grab', 'taxi', 'bus', 'fuel', 'gas'],
      Shopping: ['clothing', 'electronics', 'store', 'shopping', 'mall'],
      Utilities: ['electricity', 'water', 'internet', 'bill', 'utility'],
      Entertainment: ['movie', 'cinema', 'netflix', 'game', 'entertainment'],
    };

    let matchedCategory = 'Others';

    for (const [key, keywords] of Object.entries(categories)) {
      for (const keyword of keywords) {
        if (text.toLowerCase().includes(keyword)) {
          matchedCategory = key;
          break;
        }
      }
      if (matchedCategory !== 'Others') break;
    }

    return matchedCategory;
  };

  const handleCategorize = async () => {
    if (!image) {
      setError('No image captured.');
      return;
    }

    setLoading(true);
    setError('');
    setCategory('');
    setOcrText('');

    try {
      const preprocessedImage = await preprocessImage(image);

      const worker = await createWorker();
      await worker.load();
      await worker.reinitialize('eng');

      const { data: { text } } = await worker.recognize(preprocessedImage);

      const cleanedText = postprocessText(text);
      setOcrText(cleanedText);

      const expenseCategory = categorizeExpense(cleanedText);
      setCategory(expenseCategory);

      await worker.terminate();
    } catch (err) {
      console.error('Error during OCR processing:', err);
      setError('Failed to process the image. Please try again.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div style={styles.container}>
      <h1 style={styles.header}>Expense Scanner</h1>
      {image ? (
        <div>
          <Image
            src={image}
            alt="Captured Receipt"
            width={300}
            height={200}
            style={styles.imagePreview}
          />
        </div>
      ) : (
        <Webcam
          audio={false}
          ref={webcamRef}
          screenshotFormat="image/jpeg"
          videoConstraints={videoConstraints}
          style={styles.webcam}
        />
      )}
      <div style={styles.buttonContainer}>
        <button onClick={captureImage} style={styles.button}>
          Capture Image
        </button>
        <button onClick={handleCategorize} style={styles.button} disabled={loading || !image}>
          {loading ? 'Processing...' : 'Categorize Expense'}
        </button>
        <button
          onClick={() => setUseBackCamera((prev) => !prev)}
          style={styles.button}
        >
          Switch to {useBackCamera ? 'Front Camera' : 'Back Camera'}
        </button>
      </div>
      {error && <p style={styles.errorText}>{error}</p>}
      {ocrText && (
        <div>
          <h3 style={styles.subHeader}>OCR Text:</h3>
          <pre style={styles.ocrText}>{ocrText}</pre>
        </div>
      )}
      {category && (
        <div>
          <h3 style={styles.subHeader}>Expense Category:</h3>
          <p style={styles.categoryText}>{category}</p>
        </div>
      )}
    </div>
  );
};

  },

  },
  webcam: {
    width: '100%',
    maxWidth: '300px',
    borderRadius: '8px',
  },
  imagePreview: {
    border: '1px solid black',
    borderRadius: '8px',
  },
  buttonContainer: {
    margin: '20px',
  },
  button: {
    margin: '5px',

    backgroundColor: '#007BFF',

  },

export default Scanner;
